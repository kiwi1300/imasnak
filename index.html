using UnityEngine; using UnityEngine.SceneManagement; using System.Collections; public class GameManager : MonoBehaviour { public static GameManager Instance { get; private set; } [Header("Game States")] public bool isGameActive = false; public bool isGameOver = false; [Header("UI Elements")] public GameObject mainMenuUI; public GameObject gameOverUI; public GameObject gameplayUI; private void Awake() { // Singleton pattern to keep a single instance if (Instance == null) { Instance = this; } else { Destroy(gameObject); } } private void Start() { ShowMainMenu(); } public void StartGame() { isGameActive = true; isGameOver = false; // Toggle UI Panels if (mainMenuUI) mainMenuUI.SetActive(false); if (gameOverUI) gameOverUI.SetActive(false); if (gameplayUI) gameplayUI.SetActive(true); Time.timeScale = 1f; // Unpause the game // Trigger player or spawner initialization here if needed } public void EndGame() { isGameActive = false; isGameOver = true; if (gameplayUI) gameplayUI.SetActive(false); if (gameOverUI) gameOverUI.SetActive(true); Time.timeScale = 0f; // Pause the game action } public void RestartGame() { Time.timeScale = 1f; // Reloads the current active scene to completely clear the state SceneManager.LoadScene(SceneManager.GetActiveScene().name); } public void ShowMainMenu() { isGameActive = false; isGameOver = false; if (mainMenuUI) mainMenuUI.SetActive(true); if (gameOverUI) gameOverUI.SetActive(false); if (gameplayUI) gameplayUI.SetActive(false); Time.timeScale = 0f; // Pause until start is pressed } }
